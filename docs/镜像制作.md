# 进到自己的文件夹
cd ~/src/image

# 创建一个rootfs
mkdir rootfs

# 咱们映像的RK3588 IP 默认为 192.168.1.151
sudo rsync -avx root@192.168.1.151:/ rootfs

# 创建img文件 count是rootfs文件夹大小+1G
dd if=/dev/zero of=rk3588.img  bs=1M count=17000

# 格式化镜像文件并加入linuxroot卷标
sudo mkfs.ext4 -F -L linuxroot rk3588.img
# 考虑下方固定UUID的方式
sudo mkfs.ext4 -F -L linuxroot \
 -U a154c5c1-3bac-4f29-8bc1-6ac2898f8cde \
  rk3588.img

# 创建镜像文件挂载目录
mkdir ubuntu-mount

# 挂载空的 img 文件
sudo mount rk3588.img ubuntu-mount

# 将文件系统拷贝到 img 挂载的路径目录下
sudo cp -rfp rootfs/* ubuntu-mount

# 卸载镜像
sudo umount ubuntu-mount

# 检查并修复镜像文件系统
sudo e2fsck -p -f rk3588.img

#精简镜像文件大小
sudo resize2fs -M rk3588.img
